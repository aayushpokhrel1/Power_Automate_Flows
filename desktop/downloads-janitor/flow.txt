# =============================================================================
# Downloads Janitor  -  Power Automate Desktop paste-in  (single-loop design)
# =============================================================================
# HOW TO USE: open a new PAD flow, click the empty workspace, and paste the
# block BELOW the 8< marker. PAD reconstructs the actions.
#
# Best-effort across PAD versions. This flow uses the Contains operator and an
# If/Else-if chain that some PAD builds paste imperfectly. If anything looks
# off after pasting, recipe.md is the source of truth, build it there.
# =============================================================================

# ---- 8< ---- paste from here ---- 8< ----

System.GetEnvironmentVariable EnvironmentVariableName: $'''USERPROFILE''' EnvironmentVariableValue=> UserProfile
SET DownloadsFolder TO $'''%UserProfile%\\Downloads'''
SET DryRun TO True
SET Categorized TO 0
SET LeftAlone TO 0
SET ImagesExt TO $''';.jpg;.jpeg;.png;.gif;.webp;.svg;.bmp;.heic;.tiff;.ico;'''
SET DocumentsExt TO $''';.pdf;.doc;.docx;.txt;.md;.rtf;.odt;.epub;'''
SET SpreadsheetsExt TO $''';.xls;.xlsx;.csv;.ods;'''
SET PresentationsExt TO $''';.ppt;.pptx;.odp;'''
SET ArchivesExt TO $''';.zip;.rar;.7z;.tar;.gz;.bz2;'''
SET InstallersExt TO $''';.exe;.msi;'''
SET AudioExt TO $''';.mp3;.wav;.flac;.m4a;.aac;.ogg;'''
SET VideoExt TO $''';.mp4;.mov;.avi;.mkv;.webm;.wmv;'''
Folder.GetFiles Folder: $'''%DownloadsFolder%''' FileFilter: $'''*''' IncludeSubfolders: False FailOnAccessDenied: True Files=> Files
LOOP FOREACH CurrentFile IN Files
    Text.ChangeCase Text: $'''%CurrentFile.Extension%''' CaseType: Text.CaseType.Lowercase Result=> Ext
    SET TargetName TO $''''''
    IF ImagesExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Images'''
    ELSE IF DocumentsExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Documents'''
    ELSE IF SpreadsheetsExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Spreadsheets'''
    ELSE IF PresentationsExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Presentations'''
    ELSE IF ArchivesExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Archives'''
    ELSE IF InstallersExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Installers'''
    ELSE IF AudioExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Audio'''
    ELSE IF VideoExt Contains $''';%Ext%;''' THEN
        SET TargetName TO $'''Video'''
    END
    IF TargetName <> $'''''' THEN
        SET TargetPath TO $'''%DownloadsFolder%\\%TargetName%'''
        Folder.IfFolderExists.DoesNotExist Path: $'''%TargetPath%'''
            Folder.Create FolderPath: $'''%DownloadsFolder%''' FolderName: $'''%TargetName%''' Folder=> NewFolder
        END
        SET Categorized TO Categorized + 1
        IF DryRun = False THEN
            File.Move Files: CurrentFile Destination: $'''%TargetPath%''' IfFileExists: File.IfExists.DoNothing MovedFiles=> MovedFiles
        END
    ELSE
        SET LeftAlone TO LeftAlone + 1
    END
END
Display.ShowMessageDialog.ShowMessage Title: $'''Downloads Janitor''' Message: $'''Dry run: %DryRun%    Categorized: %Categorized%    Left alone: %LeftAlone%''' Icon: Display.Icon.None Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: False ButtonPressed=> ButtonPressed
